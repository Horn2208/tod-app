# Anmeldung und Anbieter-Protokoll — Implementierungsplan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Jeder Nutzer meldet sich selbst bei Spotify an, wodurch das Client-Secret aus der App verschwindet und der Abschnittswähler für Premium-Konten echte Wiedergabe steuert.

**Architecture:** Ein `MusicProvider`-Protokoll kapselt Anmeldung, Suche und Wiedergabe und erklärt seine Fähigkeiten; `SpotifyProvider` ist die einzige Umsetzung in diesem Plan. Die Anmeldung läuft über Authorization Code mit PKCE in einer `ASWebAuthenticationSession`, der Refresh-Token liegt im Keychain. Die Views entscheiden nach Fähigkeiten, nie nach Dienstnamen.

**Tech Stack:** Swift 6, SwiftUI, iOS 17, Swift Testing, CryptoKit, AuthenticationServices, Security (Keychain). Keine neuen Abhängigkeiten.

**Spec:** [[2026-09-04-todapp-auth-design|docs/superpowers/specs/2026-09-04-todapp-auth-design.md]]

```swift
import CryptoKit
import Foundation

/// The proof that ties an authorization code to the app that asked for it.
/// Without a client secret this is the only thing stopping someone who
/// intercepts the redirect from redeeming the code.
enum PKCE {
    private static let unreserved = Array(
        "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~"
    )

    /// 64 characters: comfortably inside RFC 7636's 43-128 window.
    static func makeVerifier(length: Int = 64) -> String {
        String((0..<length).map { _ in unreserved.randomElement()! })
    }

    static func makeState(length: Int = 32) -> String {
        String((0..<length).map { _ in unreserved.randomElement()! })
    }

    static func challenge(for verifier: String) -> String {
        let digest = SHA256.hash(data: Data(verifier.utf8))
        return base64URL(Data(digest))
    }

    /// Base64url per RFC 4648 §5: no padding, and the two substitutions that
    /// keep the value safe inside a query string.
    private static func base64URL(_ data: Data) -> String {
        data.base64EncodedString()
            .replacingOccurrences(of: "+", with: "-")
            .replacingOccurrences(of: "/", with: "_")
            .replacingOccurrences(of: "=", with: "")
    }
}
```

```swift
import Foundation

/// The URL the sign-in sheet opens, and the reading of what comes back.
/// Pure and offline, so every odd callback shape belongs in the test table.
struct SpotifyAuthorizationRequest {
    static let scopes = [
        "user-read-private",
        "user-read-playback-state",
        "user-modify-playback-state"
    ]

    let clientID: String
    let redirectURI: String
    let verifier: String
    let state: String

    var url: URL {
        var components = URLComponents(string: "https://accounts.spotify.com/authorize")!
        components.queryItems = [
            URLQueryItem(name: "response_type", value: "code"),
            URLQueryItem(name: "client_id", value: clientID),
            URLQueryItem(name: "redirect_uri", value: redirectURI),
            URLQueryItem(name: "state", value: state),
            URLQueryItem(name: "scope", value: Self.scopes.joined(separator: " ")),
            URLQueryItem(name: "code_challenge_method", value: "S256"),
            URLQueryItem(name: "code_challenge", value: PKCE.challenge(for: verifier))
        ]
        return components.url!
    }

    static func code(from callback: URL, expectedState: String) throws -> String {
        guard let components = URLComponents(url: callback, resolvingAgainstBaseURL: false) else {
            throw TodAppError.authorizationFailed
        }
        let items = Dictionary(
            (components.queryItems ?? []).compactMap { item in
                item.value.map { (item.name, $0) }
            },
            uniquingKeysWith: { first, _ in first }
        )

        if let error = items["error"] {
            // Declining is the one "error" that is really a choice.
            throw error == "access_denied"
                ? TodAppError.authorizationCancelled
                : TodAppError.authorizationFailed
        }
        // The state is checked before the code is read: a callback we did not
        // start has nothing worth reading out of it.
        guard items["state"] == expectedState, let code = items["code"] else {
            throw TodAppError.authorizationFailed
        }
        return code
    }
}
```

```swift
import Foundation

struct SpotifyTokens: Equatable, Sendable {
    var accessToken: String
    /// Nil when a refresh answered without a new one -- the old one stays valid.
    var refreshToken: String?
    var expiresAt: Date
}

/// Trades an authorization code for tokens, and later a refresh token for a
/// fresh one. No client secret is involved: with PKCE the verifier is the proof.
struct SpotifyTokenExchange: Sendable {
    typealias PerformRequest = @Sendable (URLRequest) async throws -> (Data, URLResponse)

    /// Expire a little early so a request cannot leave with a token that dies
    /// in flight -- the same margin SpotifyTokenProvider used.
    private static let safetyMargin: TimeInterval = 30

    private let clientID: String
    private let redirectURI: String
    private let performRequest: PerformRequest
    private let now: @Sendable () -> Date

    init(
        clientID: String,
        redirectURI: String,
        performRequest: @escaping PerformRequest = { try await URLSession.shared.data(for: $0) },
        now: @escaping @Sendable () -> Date = Date.init
    ) {
        self.clientID = clientID
        self.redirectURI = redirectURI
        self.performRequest = performRequest
        self.now = now
    }

    func exchange(code: String, verifier: String) async throws -> SpotifyTokens {
        try await post([
            "grant_type": "authorization_code",
            "code": code,
            "redirect_uri": redirectURI,
            "client_id": clientID,
            "code_verifier": verifier
        ])
    }

    func refresh(_ refreshToken: String) async throws -> SpotifyTokens {
        try await post([
            "grant_type": "refresh_token",
            "refresh_token": refreshToken,
            "client_id": clientID
        ])
    }

    private struct Response: Decodable {
        let access_token: String
        let refresh_token: String?
        let expires_in: Double?
    }

    private func post(_ fields: [String: String]) async throws -> SpotifyTokens {
        var request = URLRequest(url: URL(string: "https://accounts.spotify.com/api/token")!)
        request.httpMethod = "POST"
        request.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
        var body = URLComponents()
        body.queryItems = fields.map { URLQueryItem(name: $0.key, value: $0.value) }
        request.httpBody = Data((body.percentEncodedQuery ?? "").utf8)

        let (data, response) = try await performRequest(request)
        guard let http = response as? HTTPURLResponse, http.statusCode == 200,
              let decoded = try? JSONDecoder().decode(Response.self, from: data) else {
            throw TodAppError.authorizationFailed
        }
        let lifetime = decoded.expires_in ?? 3600
        return SpotifyTokens(
            accessToken: decoded.access_token,
            refreshToken: decoded.refresh_token,
            expiresAt: now().addingTimeInterval(max(0, lifetime - Self.safetyMargin))
        )
    }
}
```

```swift
import Foundation
import Security

/// Where the refresh token lives between launches. The access token is
/// deliberately not stored: it is short-lived and cheap to fetch again.
protocol TokenStore: Sendable {
    func read() throws -> String?
    func write(_ refreshToken: String) throws
    func clear() throws
}

/// The Keychain, so the token survives a reinstall of nothing and is not
/// readable from a file backup the way UserDefaults would be.
struct KeychainTokenStore: TokenStore {
    let service: String
    let account: String

    init(service: String = "com.lennetschierhorn.TodApp", account: String = "spotify-refresh-token") {
        self.service = service
        self.account = account
    }

    private var query: [String: Any] {
        [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: account
        ]
    }

    func read() throws -> String? {
        var lookup = query
        lookup[kSecReturnData as String] = true
        lookup[kSecMatchLimit as String] = kSecMatchLimitOne

        var item: CFTypeRef?
        let status = SecItemCopyMatching(lookup as CFDictionary, &item)
        if status == errSecItemNotFound { return nil }
        guard status == errSecSuccess, let data = item as? Data else {
            throw TodAppError.authorizationFailed
        }
        return String(decoding: data, as: UTF8.self)
    }

    func write(_ refreshToken: String) throws {
        // Delete first: SecItemAdd on an existing item fails with
        // errSecDuplicateItem, and an update needs a different call shape.
        SecItemDelete(query as CFDictionary)
        var item = query
        item[kSecValueData as String] = Data(refreshToken.utf8)
        item[kSecAttrAccessible as String] = kSecAttrAccessibleAfterFirstUnlock
        guard SecItemAdd(item as CFDictionary, nil) == errSecSuccess else {
            throw TodAppError.authorizationFailed
        }
    }

    func clear() throws {
        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw TodAppError.authorizationFailed
        }
    }
}

/// Stands in for the Keychain in tests, which have no entitlements.
final class InMemoryTokenStore: TokenStore, @unchecked Sendable {
    private let lock = NSLock()
    private var stored: String?

    init(stored: String? = nil) {
        self.stored = stored
    }

    func read() throws -> String? {
        lock.lock(); defer { lock.unlock() }
        return stored
    }

    func write(_ refreshToken: String) throws {
        lock.lock(); defer { lock.unlock() }
        stored = refreshToken
    }

    func clear() throws {
        lock.lock(); defer { lock.unlock() }
        stored = nil
    }
}
```

```swift
/// Spotify behind the provider door. An actor because the token and the
/// capabilities are mutable state that several screens read at once.
actor SpotifyProvider: MusicProvider {
    typealias PerformRequest = @Sendable (URLRequest) async throws -> (Data, URLResponse)
    /// Opens the sign-in sheet and returns the callback URL. Injected so the
    /// tests never touch AuthenticationServices.
    typealias PresentSignIn = @Sendable (URL, String) async throws -> URL

    nonisolated let service: MusicService = .spotify

    private let clientID: String
    private let redirectURI: String
    private let tokenStore: any TokenStore
    private let exchange: SpotifyTokenExchange
    private let performRequest: PerformRequest
    private let presentSignIn: PresentSignIn
    private let now: @Sendable () -> Date

    private var tokens: SpotifyTokens?
    private var declaredCapabilities: MusicProviderCapabilities = .none

    init(
        clientID: String,
        redirectURI: String = "todapp://auth/spotify",
        tokenStore: any TokenStore,
        exchange: SpotifyTokenExchange,
        performRequest: @escaping PerformRequest = { try await URLSession.shared.data(for: $0) },
        presentSignIn: @escaping PresentSignIn,
        now: @escaping @Sendable () -> Date = Date.init
    ) {
        self.clientID = clientID
        self.redirectURI = redirectURI
        self.tokenStore = tokenStore
        self.exchange = exchange
        self.performRequest = performRequest
        self.presentSignIn = presentSignIn
        self.now = now
    }

    var isAuthorized: Bool { tokens != nil }
    var capabilities: MusicProviderCapabilities { declaredCapabilities }

    func authorize() async throws {
        let verifier = PKCE.makeVerifier()
        let state = PKCE.makeState()
        let request = SpotifyAuthorizationRequest(
            clientID: clientID, redirectURI: redirectURI, verifier: verifier, state: state
        )

        let callback = try await presentSignIn(request.url, scheme(of: redirectURI))
        let code = try SpotifyAuthorizationRequest.code(from: callback, expectedState: state)

        let fresh = try await exchange.exchange(code: code, verifier: verifier)
        try store(fresh)
        await loadCapabilities()
    }

    /// Brings a previous session back without asking again. A refresh that
    /// fails simply means signed out -- there is nothing to report to anyone.
    func restore() async {
        guard let refreshToken = try? tokenStore.read() else { return }
        guard let fresh = try? await exchange.refresh(refreshToken) else {
            try? tokenStore.clear()
            return
        }
        try? store(fresh, fallbackRefreshToken: refreshToken)
        await loadCapabilities()
    }

    func signOut() async {
        tokens = nil
        declaredCapabilities = .none
        try? tokenStore.clear()
    }

    func searchTracks(matching query: String) async throws -> [TrackCandidate] {
        // Filled in by Task 7.
        []
    }

    func play(_ source: TrackSource, from section: TrackSection?) async throws {
        // Filled in by Task 8.
        throw TodAppError.playbackNotAvailable
    }

    func stop() async {}

    // MARK: - Innenleben

    private struct Profile: Decodable { let product: String? }

    /// Reads `product` to decide what this account may do. An unreadable
    /// profile grants nothing: guessing generously here produces a button that
    /// answers 403.
    private func loadCapabilities() async {
        guard let token = tokens?.accessToken else {
            declaredCapabilities = .none
            return
        }
        var request = URLRequest(url: URL(string: "https://api.spotify.com/v1/me")!)
        request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")

        var premium = false
        if let (data, response) = try? await performRequest(request),
           let http = response as? HTTPURLResponse, http.statusCode == 200,
           let profile = try? JSONDecoder().decode(Profile.self, from: data) {
            premium = profile.product == "premium"
        }
        declaredCapabilities = MusicProviderCapabilities(
            canSearch: true,
            canPlayFullTracks: premium,
            canStartAtPosition: premium,
            playsInsideTheApp: false
        )
    }

    private func store(_ fresh: SpotifyTokens, fallbackRefreshToken: String? = nil) throws {
        tokens = fresh
        // A refresh may answer without a new refresh token; the old one keeps
        // working, and overwriting it with nothing would sign the user out.
        if let refreshToken = fresh.refreshToken ?? fallbackRefreshToken {
            try tokenStore.write(refreshToken)
        }
    }

    private func scheme(of uri: String) -> String {
        URLComponents(string: uri)?.scheme ?? "todapp"
    }
}
```

```swift
        let trimmed = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { return [] }

        var components = URLComponents(string: "https://api.spotify.com/v1/search")!
        components.queryItems = [
            URLQueryItem(name: "q", value: trimmed),
            URLQueryItem(name: "type", value: "track"),
            // A Development Mode app registration refuses any limit above 10.
            URLQueryItem(name: "limit", value: "10")
        ]
        var request = URLRequest(url: components.url!)
        request.setValue("Bearer \(try await validToken())", forHTTPHeaderField: "Authorization")

        let (data, response) = try await performRequest(request)
        guard let http = response as? HTTPURLResponse else { throw TodAppError.searchUnavailable }
        switch http.statusCode {
        case 200: return try SpotifyResponseMapper.candidates(from: data)
        case 429: throw TodAppError.rateLimited
        // The app holds no credentials of its own any more, so a refused token
        // means this person needs to sign in again.
        case 401, 403: throw TodAppError.authorizationFailed
        default: throw TodAppError.trackUnavailable
        }

    /// Renews an expired token before it is used, so a long session does not
    /// start failing halfway through. Several screens can ask at once on a
    /// cold token; the first caller's work is what the others await, or
    /// Spotify sees a burst of identical refreshes and may invalidate them.
    private func validToken() async throws -> String {
        if let current = tokens, current.expiresAt > now() { return current.accessToken }
        if let refreshInFlight { return try await refreshInFlight.value }

        let task = Task<String, Error> { [exchange, tokenStore] in
            guard let refreshToken = try? tokenStore.read(),
                  let fresh = try? await exchange.refresh(refreshToken) else {
                throw TodAppError.authorizationFailed
            }
            return try await self.adopt(fresh, fallbackRefreshToken: refreshToken)
        }
        refreshInFlight = task
        defer { refreshInFlight = nil }
        return try await task.value
    }

    private func adopt(_ fresh: SpotifyTokens, fallbackRefreshToken: String) throws -> String {
        try store(fresh, fallbackRefreshToken: fallbackRefreshToken)
        return fresh.accessToken
```

```swift
        // PREMIUM_REQUIRED, and a button that only fails is worse than one
```

```swift
/// The only place that touches AuthenticationServices, so everything else can
/// be tested without a sheet appearing.
enum WebAuthentication {
    @MainActor
    static func presentSignIn(url: URL, callbackScheme: String) async throws -> URL {
        try await withCheckedThrowingContinuation { continuation in
            let session = ASWebAuthenticationSession(
                url: url, callbackURLScheme: callbackScheme
            ) { callback, error in
                if let callback {
                    continuation.resume(returning: callback)
                } else if let error = error as? ASWebAuthenticationSessionError,
                          error.code == .canceledLogin {
                    continuation.resume(throwing: TodAppError.authorizationCancelled)
                } else {
                    continuation.resume(throwing: TodAppError.authorizationFailed)
                }
            }
            session.presentationContextProvider = PresentationAnchor.shared
            // Keeps the Spotify session out of Safari's cookie jar and makes a
            // second sign-in ask again rather than silently reusing an account.
            session.prefersEphemeralWebBrowserSession = true
            session.start()
        }
    }

    private final class PresentationAnchor: NSObject, ASWebAuthenticationPresentationContextProviding {
        static let shared = PresentationAnchor()

        func presentationAnchor(for session: ASWebAuthenticationSession) -> ASPresentationAnchor {
            ASPresentationAnchor()
        }
```
